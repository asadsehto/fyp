# faster-whisper: runs Whisper large-v3-turbo on CPU efficiently
# Uses CTranslate2 under the hood — 4x faster than original whisper
# First run downloads the model (~800 MB) automatically
faster-whisper==1.1.1

# argostranslate: 100% offline Urdu → English translation
# Downloads a ~50 MB language pack once, then works forever without internet
argostranslate==1.9.6

# sounddevice: records audio from the laptop microphone
# Wraps PortAudio — works on Windows, Mac, Linux
sounddevice==0.5.1

# numpy: handles audio as arrays of numbers
numpy>=1.24.0

# scipy: saves temporary WAV files from numpy arrays
scipy>=1.10.0
